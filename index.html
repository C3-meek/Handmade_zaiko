ハンドメイド在庫帳

作った商品の数と、元になる材料の残りを一覧で確認できます。

初回のみ（画像機能の追加時も）：Supabase の SQL Editor で実行するSQL 
create table if not exists materials (
  id bigint generated always as identity primary key,
  code text not null unique,      -- 管理番号2（材料）
  name text not null,
  level text not null default 'ok' check (level in ('ok','low','none')),
  created_at timestamptz default now()
);
create table if not exists products (
  id bigint generated always as identity primary key,
  code text not null unique,      -- 管理番号（商品）
  name text not null,
  qty int not null default 0,
  material_id bigint references materials(id) on delete set null,
  created_at timestamptz default now()
);
-- 画像用の列（前回のSQLを実行済みの方もここから実行できます）
alter table materials add column if not exists image_url text;
alter table products add column if not exists image_url text;

alter table materials enable row level security;
alter table products enable row level security;
drop policy if exists "all materials" on materials;
drop policy if exists "all products" on products;
create policy "all materials" on materials for all using (true) with check (true);
create policy "all products" on products for all using (true) with check (true);

-- 画像の保存先（公開バケット）
insert into storage.buckets (id, name, public) values ('images_zaiko','images_zaiko',true)
  on conflict (id) do nothing;
drop policy if exists "images_zaiko read" on storage.objects;
drop policy if exists "images_zaiko write" on storage.objects;
create policy "images_zaiko read" on storage.objects for select using (bucket_id = 'images_zaiko');
create policy "images_zaiko write" on storage.objects for insert with check (bucket_id = 'images_zaiko');

このSQLは、キーを知っている人なら誰でも読み書き・画像の追加ができる設定です。画像は公開URLになります。個人用の簡易版として使い、公開リポジトリにキーを書き込まないでください（キーはこのブラウザにだけ保存されます）。
